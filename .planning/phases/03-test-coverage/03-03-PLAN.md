    void setUp() throws Exception {
        // Create MyBatis-Plus SqlSessionFactory with the DataSource
        // Register SubmissionMapper, UserMapper, ProblemMapper

        MybatisConfiguration configuration = new MybatisConfiguration();

        configuration.addMapper(UserMapper.class);
        configuration.addMapper(ProblemMapper.class);
        // Set up MybatisSqlSessionFactoryBean with dataSource and configuration
