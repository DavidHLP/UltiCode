   ```java
   @Override
   public PageResult<AdminForumController.AdminForumCommunityVO> getCommunities(int page, int limit) {
       Page<ForumCommunity> pageResult = new Page<>(page, limit);
       Page<ForumCommunity> result = forumCommunityMapper.selectPage(pageResult,
           new LambdaQueryWrapper<ForumCommunity>().orderByDesc(ForumCommunity::getMembers));
       List<AdminForumController.AdminForumCommunityVO> voList = result.getRecords().stream()
           .map(c -> {
               AdminForumController.AdminForumCommunityVO vo = new AdminForumController.AdminForumCommunityVO();
               vo.setId(c.getId());
               vo.setName(c.getName());
               vo.setSlug(c.getSlug());
               vo.setDescription(c.getDescription());
               vo.setPostCount(c.getPostsCount() != null ? c.getPostsCount() : 0);
               vo.setMemberCount(c.getMembers() != null ? c.getMembers() : 0);
               return vo;
           })
           .collect(java.util.stream.Collectors.toList());
       return PageResult.of(voList, result.getTotal(), page, limit);
   }
   ```
